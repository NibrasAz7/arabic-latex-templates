\section{المادة الخامسة: الملكية الفكرية (\LR{Intellectual Property})}
\label{sec:ip}
\begin{enumerate}
    \item تنتقل جميع حقوق الملكية الفكرية (\LR{IP Rights}) الناتجة عن الخدمات إلى العميل عند سداد الدفعات كاملةً.
    \item يحتفظ مقدّم الخدمة بالأدوات والمنهجيات العامة (\LR{Pre-existing Tools}) التي استُخدمت قبل الاتفاقية.
    \item يضمن مقدّم الخدمة أن الخدمات لا تنتهك حقوق ملكية فكرية لأطراف ثالثة.
\end{enumerate}

% ============================================================
% 6. CONFIDENTIALITY
% ============================================================
\section{المادة السادسة: السرية (\LR{Confidentiality})}
\label{sec:confidentiality}
\begin{enumerate}
    \item يلتزم مقدّم الخدمة بالحفاظ على سرية (\LR{Confidentiality}) جميع معلومات العميل.
    \item لا يُفشِي المعلومات لأي طرف ثالث إلا بموافقة العميل كتابةً.
    \item يستمر الالتزام بالسرية لمدة \rule{2cm}{0.4pt} سنوات بعد انتهاء الاتفاقية.
\end{enumerate}

% ============================================================
% 7. WARRANTIES
% ============================================================
\section{المادة السابعة: الضمانات (\LR{Warranties})}
\label{sec:warranties}
\begin{enumerate}
    \item \textbf{جودة الخدمات:} يضمن مقدّم الخدمة أن الخدمات تُقدَّم بمستوى مهني عالٍ.
    \item \textbf{فترة الضمان:} تُصلَح العيوب (\LR{Defects}) مجاناً خلال \rule{2cm}{0.4pt} يوماً من التسليم.
    \item \textbf{الاستثناءات:} لا يشمل الضمان الأضرار الناتجة عن إساءة استخدام العميل أو تعديلات خارجية.
\end{enumerate}

% ============================================================
% 8. TERMINATION
% ============================================================
\section{المادة الثامنة: الإنهاء (\LR{Termination})}
\label{sec:termination}
\begin{enumerate}
    \item \textbf{الإنهاء بإشعار:} يجوز لأي طرف إنهاء الاتفاقية بإشعار مسبق \rule{2cm}{0.4pt} يوماً.
    \item \textbf{الإنهاء للسبب:} يجوز الإنهاء فوراً عند الإخلال الجسيم، بعد منح مهلة \rule{2cm}{0.4pt} يوماً للتصحيح.
    \item \textbf{آثار الإنهاء:} يحق للعميل استرداد الدفعات المقدّمة مقابل الخدمات غير المُقدَّمة، ويلتزم مقدّم الخدمة بتسليم ما أنجزه.
\end{enumerate}
