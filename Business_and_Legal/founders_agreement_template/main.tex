% 4. VESTING
% ============================================================
\section{جدول الاستحقاق (\LR{Vesting Schedule})}
\label{sec:vesting}

\begin{enumerate}
    \item تُستحق الأسهم على مدى \textbf{4 سنوات} من تاريخ بدء العمل.
    \item \textbf{فترة الجرف (\LR{Cliff}):} لا تُستحق أي أسهم خلال السنة الأولى. عند إكمال السنة الأولى، تُستحق \textbf{25\%} من الأسهم دفعة واحدة.
    \item بعد فترة الجرف، تُستحق الأسهم المتبقية شهرياً على مدى \textbf{36 شهراً}.
    \item \textbf{الاستحقاق المُسرّع:} عند الاستحواذ على الشركة، تُستحق جميع الأسهم فوراً (\LR{Single-trigger}) أو عند الاستحواذ والإنهاء معاً (\LR{Double-trigger}).
\end{enumerate}

\section{اتخاذ القرارات}
\label{sec:decisions}

\subsection{القرارات العادية}
تُتخذ بالأغلبية المطلقة (أكثر من 50\% من الأصوات).

\subsection{القرارات الجوهرية}
تتطلب موافقة \rule{2cm}{0.4pt}\% من الأصوات، وتشمل:
\begin{enumerate}
    \item بيع الشركة أو الاستحواذ عليها.
    \item إصدار أسهم جديدة.
    \item تعديل اتفاقية المؤسسين.
    \item تعيين أو عزل مدير تنفيذي.
    \item الموافقة على الميزانية السنوية.
    \item الحصول على قروض تتجاوز \rule{3cm}{0.4pt} \LR{USD}.
\end{enumerate}

\section{الملكية الفكرية}
\label{sec:ip}

\begin{enumerate}
    \item تنتقل جميع حقوق الملكية الفكرية التي يُنشئها أي مؤسس (قبل أو بعد التأسيس) والمتعلقة بالشركة إلى ملكية الشركة كاملة.
    \item يُوقّع كل مؤسس اتفاقية تنازل منفصلة عن الملكية الفكرية.
    \item لا يجوز لأي مؤسس استخدام الملكية الفكرية لأغراض شخصية أو منافسة.
\end{enumerate}

% ============================================================
% 8. CONFIDENTIALITY
% ============================================================
\section{السرية وعدم المنافسة}
\label{sec:confidentiality}

\begin{enumerate}
    \item \textbf{السرية:} يلتزم كل مؤسس بالحفاظ على سرية معلومات الشركة وعدم إفشائها.
    \item \textbf{عدم المنافسة:} لا يجوز لأي مؤسس تأسيس أو العمل في شركة منافسة خلال فترة عمله في الشركة ولمدة \rule{2cm}{0.4pt} شهراً بعد مغادرته.
    \item \textbf{عدم الاستمالة:} لا يجوز لأي مؤسس استمالة موظفي الشركة أو عملائها بعد مغادرته لمدة \rule{2cm}{0.4pt} شهراً.
\end{enumerate}

% ============================================================
% 9. DEPARTURE
% ============================================================
\section{مغادرة المؤسس}
\label{sec:departure}

\subsection{المغادرة الطوعية}
\begin{enumerate}
    \item يُشترط إشعار مسبق بمدة \rule{2cm}{0.4pt} يوماً.
    \item تُستحق الأسهم وفقاً لجدول الاستحقاق فقط.
    \item تُعاد الأسهم غير المستحقة إلى الشركة.
\end{enumerate}

\subsection{الإنهاء لسبب (\LR{For Cause})}
\begin{enumerate}
    \item يجوز إنهاء المؤسس لسبب (انتهاك الاتفاقية، إهمال جسيم، احتيال).
    \item في هذه الحالة، تُستحق الأسهم المستحقة فقط، وتُعاد الباقية.
\end{enumerate}

\subsection{الوفاة أو العجز}
\begin{enumerate}
    \item تنتقل الأسهم المستحقة إلى الورثة.
    \item تُعاد الأسهم غير المستحقة إلى الشركة.
    \item يحق للشركة شراء الأسهم من الورثة بالقيمة العادلة.
\end{enumerate}

\end{enumerate}
