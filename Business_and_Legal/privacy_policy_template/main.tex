\section{استخدام البيانات (\LR{Use of Data})}
\label{sec:use}
نستخدم بياناتك الشخصية للأغراض التالية:
\begin{enumerate}
    \item \textbf{تقديم الخدمات:} تنفيذ طلباتك وتقديم الخدمات المطلوبة.
    \item \textbf{تحسين الخدمات:} تحليل الاستخدام وتطوير ميزات جديدة.
    \item \textbf{التواصل:} الرد على استفساراتك وإرسال الإشعارات.
    \item \textbf{الامتثال القانوني:} الالتزام بالالتزامات القانونية والتنظيمية.
    \item \textbf{الأمن:} منع الاحتيال وحماية الحسابات.
\end{enumerate}

% ============================================================
% 4. DATA SHARING
% ============================================================
\section{مشاركة البيانات (\LR{Data Sharing})}
\label{sec:sharing}
لا نبيع بياناتك الشخصية. قد نشاركها في الحالات التالية:
\begin{enumerate}
    \item \textbf{مزوّدو الخدمات (\LR{Service Providers}):} أطراف ثالثة تساعدنا في تقديم خدماتنا، مع التزامهم بسرية البيانات.
    \item \textbf{السلطات القانونية (\LR{Legal Authorities}):} عند الطلب القانوني أو أمر قضائي.
    \item \textbf{تحويلات الأعمال (\LR{Business Transfers}):} في حال الاندماج أو الاستحواذ، مع التزام الطرف المستحوِظ بنفس المعايير.
    \item \textbf{موافقتك (\LR{With Consent}):} عند موافقتك الصريحة على المشاركة.
\end{enumerate}

% ============================================================
% 5. DATA SECURITY
% ============================================================
\section{أمن البيانات (\LR{Data Security})}
\label{sec:security}
نتخذ تدابير أمنية مناسبة لحماية بياناتك، تشمل:
\begin{enumerate}
    \item التشفير (\LR{Encryption}) أثناء النقل والتخزين.
    \item ضوابط الوصول (\LR{Access Controls}) المحدودة للموظفين المصرّح لهم.
    \item مراجعات أمنية دورية وتقييم المخاطر.
    \item خطة استجابة للحوادث (\LR{Incident Response Plan}) في حال حدوث خرق.
\end{enumerate}

\noindent\textbf{الإخطار بالخرق:} في حال حدوث خرق للبيانات (\LR{Data Breach})، نلتزم بإخطار المستخدمين والجهات المختصة خلال \rule{2cm}{0.4pt} ساعة وفقاً لمتطلبات \LR{GDPR}.

\section{حقوق المستخدم (\LR{User Rights})}
\label{sec:rights}
يتمتع المستخدمون بالحقوق التالية وفقاً لـ \LR{GDPR} و \LR{CCPA}:
\begin{enumerate}
    \item \textbf{حق الوصول (\LR{Right of Access}):} طلب نسخة من بياناتك الشخصية.
    \item \textbf{حق التصحيح (\LR{Right to Rectification}):} تصحيح البيانات غير الدقيقة.
    \item \textbf{حق الحذف (\LR{Right to Erasure}):} طلب حذف بياناتك (حق النسيان).
    \item \textbf{حق تقييد المعالجة (\LR{Right to Restriction}):} تقييد معالجة بياناتك في حالات معينة.
    \item \textbf{حق نقل البيانات (\LR{Right to Data Portability}):} استلام بياناتك بصيغة قابلة للنقل.
    \item \textbf{حق الاعتراض (\LR{Right to Object}):} الاعتراض على معالجة بياناتك.
    \item \textbf{حق سحب الموافقة (\LR{Right to Withdraw Consent}):} سحب موافقتك في أي وقت.
\end{enumerate}

\noindent لممارسة أي من هذه الحقوق، يُرجى التواصل معنا عبر القسم الثامن.
