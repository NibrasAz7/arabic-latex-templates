% ============================================================
% 3. DUTIES
% ============================================================
\section{الواجبات والمسؤوليات}
\label{sec:duties}
\begin{enumerate}
    \item أداء المهام الموكلة إليه بأمانة وكفاءة.
    \item الالتزام بأنظمة وسياسات الشركة.
    \item الحفاظ على سرية معلومات الشركة.
    \item عدم العمل لدى طرف آخر دون موافقة الشركة.
    \item تقديم تقارير دورية عن أدائه.
\end{enumerate}

    \item \textbf{ساعات العمل اليومية:} من \rule{2cm}{0.4pt} إلى \rule{2cm}{0.4pt}.
    \item يجوز طلب العمل الإضافي عند الحاجة، ويُعوَّض وفقاً للقانون.
\end{enumerate}

% ============================================================
% 7. CONFIDENTIALITY
% ============================================================
\section{السرية وعدم المنافسة}
\label{sec:confidentiality}
\begin{enumerate}
    \item \textbf{السرية:} يلتزم الموظف بالحفاظ على سرية معلومات الشركة.
    \item \textbf{عدم المنافسة:} لا يجوز للموظف العمل لدى منافس لمدة \rule{2cm}{0.4pt} شهراً بعد انتهاء العقد.
    \item \textbf{عدم استمالة الموظفين:} لا يجوز استمالة موظفي الشركة لمدة \rule{2cm}{0.4pt} شهراً بعد انتهاء العقد.
    \item \textbf{الملكية الفكرية:} تنتقل جميع حقوق الملكية الفكرية التي يُنشئها الموظف إلى الشركة.
\end{enumerate}

% 8. TERMINATION
% ============================================================
\section{إنهاء العقد}
\label{sec:termination}
\subsection{إنهاء من قبل الشركة}
\begin{enumerate}
    \item يجوز للشركة إنهاء العقد بإشعار مسبق \rule{2cm}{0.4pt} يوماً، مع تعويض وفقاً للقانون.
    \item يجوز الإنهاء فوراً «لسبب» (إخلال جسيم، إهمال، احتيال) دون تعويض.
\end{enumerate}

\subsection{إنهاء من قبل الموظف}
\begin{enumerate}
    \item يلتزم الموظف بإشعار مسبق \rule{2cm}{0.4pt} يوماً.
    \item يلتزم بتسليم جميع ممتلكات الشركة ووثائقها.
\end{enumerate}
