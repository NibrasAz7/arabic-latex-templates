% ============================================================
% 1. BACKGROUND
% ============================================================
\section{الخلفية}
\label{sec:background}
يعمل/يؤسس المُتنازِل لدى الشركة بصفته \rule{6cm}{0.4pt}، وقد يقوم بابتكار أو تطوير أو المساهمة في ملكيات فكرية خلال فترة عمله. يوافق المُتنازِل بموجب هذه الاتفاقية على تنازل جميع حقوقه في هذه الملكيات للشركة.

% ============================================================
% 2. DEFINITION OF IP
% ============================================================
\section{تعريف الملكية الفكرية}
\label{sec:definition}
تشمل «الملكية الفكرية» جميع الإبداعات الفكرية، بما في ذلك:
\begin{enumerate}
    \item البرمجيات والكود المصدري والكود المُجمَّع.
    \item التصاميم والواجهات والرسومات.
    \item براءات الاختراع ونماذج المنفعة.
    \item العلامات التجارية والأسماء التجارية والشعارات.
    \item حقوق النشر والتأليف.
    \item الأسرار التجارية والمعرفة الفنية (\LR{Know-how}).
    \item قواعد البيانات والمحتوى.
    \item أي إبداع قابل للحماية القانونية.
\end{enumerate}

% ============================================================
% 3. ASSIGNMENT
% ============================================================
\section{التنازل عن الملكية الفكرية}
\label{sec:assignment}
\begin{enumerate}
    \item يتنازل المُتنازِل بموجب هذه الاتفاقية عن جميع حقوقه في الملكية الفكرية التي يُنشئها أو يساهم في إنشائها خلال فترة عمله لدى الشركة، سواء كانت قبل أو بعد تاريخ هذه الاتفاقية.
    \item يشمل التنازل جميع الحقوق المالية والمعنوية (بما في ذلك حق النشر والتوزيع والتعديل والترخيص).
    \item يكون التنازل شاملاً لجميع دول العالم ولمدة غير محدودة.
    \item يوافق المُتنازِل على توقيع أي وثائق إضافية قد تطلبها الشركة لتسجيل أو حماية هذه الحقوق.
\end{enumerate}

% ============================================================
% 4. PRIOR IP
% ============================================================
\section{الملكية الفكرية السابقة}
\label{sec:prior}
\begin{enumerate}
    \item يُرفق المُتنازِل قائمة بالملكية الفكرية التي يملكها قبل تاريخ هذه الاتفاقية (الملاحق).
    \item لا تشمل هذه الاتفاقية الملكية الفكرية السابقة المذكورة في القائمة.
    \item إذا استُخدمت الملكية الفكرية السابقة في أعمال الشركة، تمنح الشركة ترخيصاً غير حصري وغير قابل للتحويل لاستخدامها.
\end{enumerate}

% ============================================================
% 5. WORK MADE FOR HIRE
% ============================================================
\section{أعمال مُنجزة لصالح الشركة}
\label{sec:workmade}
\begin{enumerate}
    \item تُعتبر جميع الأعمال التي يُنجزها المُتنازِل خلال فترة عمله «أعمالاً مُنجزة لصالح الشركة» (\LR{Work Made for Hire}).
    \item تنتقل جميع حقوق الملكية فيها إلى الشركة فور إنشائها.
    \item يلتزم المُتنازِل بإبلاغ الشركة عن أي عمل قابل للحماية فور إنشائه.
\end{enumerate}

% ============================================================
% 6. ASSISTANCE
% ============================================================
\section{المساعدة في حماية الحقوق}
\label{sec:assistance}
\begin{enumerate}
    \item يلتزم المُتنازِل بمساعدة الشركة في تسجيل أو حماية الملكية الفكرية.
    \item يوقّع جميع الوثائق اللازمة لتسجيل براءات الاختراع أو العلامات التجارية.
    \item يبلغ الشركة عن أي انتهاك محتمل للملكية الفكرية.
    \item يستمر هذا الالتزام حتى بعد انتهاء العلاقة التعاقدية.
\end{enumerate}

% ============================================================
% 7. NO CONFLICT
% ============================================================
\section{عدم تعارض المصالح}
\label{sec:noconflict}
\begin{enumerate}
    \item يقرّ المُتنازِل بأنه لا يوجد أي التزام يتعارض مع هذه الاتفاقية.
    \item لا يوجد اتفاق سابق يمنحه حقوقاً في الملكية الفكرية موضوع هذه الاتفاقية.
    \item يتحمل المُتنازِل المسؤولية عن أي ادعاءات من أطراف ثالثة بشأن الملكية الفكرية.
\end{enumerate}

% ============================================================
% 8. CONFIDENTIALITY
% ============================================================
\section{السرية}
\label{sec:confidentiality}
يلتزم المُتنازِل بالحفاظ على سرية جميع معلومات الشركة والملكية الفكرية، وفقاً لاتفاقية عدم الإفصاح (\LR{NDA}) الموقّعة بين الطرفين.
