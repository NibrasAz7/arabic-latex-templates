\section{الجمعية العامة للمساهمين}
\label{ch:shareholders}

\subsection{الاجتماعات}
\begin{enumerate}
    \item تُعقد الجمعية العامة العادية مرة واحدة سنوياً خلال \rule{2cm}{0.4pt} أشهر من نهاية السنة المالية.
    \item تُعقد الجمعية العامة غير العادية بناءً على طلب مجلس الإدارة أو \rule{2cm}{0.4pt}\% من المساهمين.
    \item يُرسل إشعار الاجتماع قبل \rule{2cm}{0.4pt} يوماً على الأقل.
\end{enumerate}

\subsection{النصاب والتصويت}
\begin{enumerate}
    \item يُعتبر الاجتماع قانونياً بحضور \rule{2cm}{0.4pt}\% من المساهمين.
    \item تُتخذ القرارات بالأغلبية المطلقة ما لم يُنص على خلاف ذلك.
    \item لكل سهم صوت واحد في التصويت.
\end{enumerate}

\subsection{صلاحيات الجمعية العامة}
\begin{enumerate}
    \item اعتماد القوائم المالية السنوية.
    \item انتخاب أعضاء مجلس الإدارة.
    \item تعيين المراجع المالي.
    \item توزيع الأرباح.
    \item تعديل النظام الأساسي.
    \item حلّ الشركة.
\end{enumerate}

% CHAPTER 3: BOARD OF DIRECTORS
% ============================================================
\section{مجلس الإدارة}
\label{ch:board}

\subsection{التشكيل}
\begin{enumerate}
    \item يتألف مجلس الإدارة من \rule{2cm}{0.4pt} عضو يُنتخبون من قبل الجمعية العامة.
    \item مدة عضوية المجلس \rule{2cm}{0.4pt} سنوات قابلة للتجديد.
    \item يُنتخب رئيس مجلس الإدارة من بين الأعضاء.
\end{enumerate}

\subsection{الاجتماعات}
\begin{enumerate}
    \item يُعقد المجلس \rule{2cm}{0.4pt} مرات سنوياً على الأقل.
    \item النصاب القانوني: \rule{2cm}{0.4pt}\% من الأعضاء.
    \item تُتخذ القرارات بالأغلبية، وفي حالة التساوي يكون لرئيس المجلس صوت مرجّح.
\end{enumerate}

\subsection{صلاحيات المجلس}
\begin{enumerate}
    \item وضع السياسات العامة للشركة.
    \item تعيين المدير التنفيذي.
    \item اعتماد الميزانية السنوية.
    \item الموافقة على القروض والاستثمارات الكبرى.
    \item تعيين كبار الموظفين.
\end{enumerate}

\section{المديرون التنفيذيون}
\label{ch:officers}

\subsection{المناصب}
يتضمن الهيكل التنفيذي للشركة المناصب التالية:
\begin{enumerate}
    \item \textbf{المدير التنفيذي (\LR{CEO}):} المسؤول عن الإدارة العامة.
    \item \textbf{المدير المالي (\LR{CFO}):} المسؤول عن الشؤون المالية.
    \item \textbf{المدير التقني (\LR{CTO}):} المسؤول عن الشؤون التقنية.
    \item \textbf{أمين السر (\LR{Secretary}):} المسؤول عن الوثائق والمراسلات.
\end{enumerate}

\subsection{التعيين والعزل}
\begin{enumerate}
    \item يُعيّن المديرون التنفيذيون بقرار من مجلس الإدارة.
    \item يُعزلون بقرار من المجلس بالأغلبية.
\end{enumerate}

\section{الأحكام المالية}
\label{ch:finance}

\subsection{السنة المالية}
تبدأ السنة المالية في \rule{3cm}{0.4pt} وتنتهي في \rule{3cm}{0.4pt}.

\subsection{القوائم المالية}
\begin{enumerate}
    \item تُعدّ القوائم المالية وفقاً لمعايير \rule{4cm}{0.4pt}.
    \item يُعين مراجع مالي مستقل من قبل الجمعية العامة.
    \item تُحفظ الدفاتر المحاسبية في المقر الرئيسي.
\end{enumerate}

\subsection{توزيع الأرباح}
\begin{enumerate}
    \item يُحدد مجلس الإدارة نسبة الأرباح القابلة للتوزيع.
    \item يُخصم \rule{2cm}{0.4pt}\% كاحتياطي قانوني قبل التوزيع.
    \item تُوزّع الأرباح المتبقية على المساهمين بنسبة حصصهم.
\end{enumerate}

% CHAPTER 6: TRANSFER OF SHARES
% ============================================================
\section{نقل الأسهم}
\label{ch:transfer}

\begin{enumerate}
    \item لا يجوز نقل الأسهم إلا بموافقة \rule{2cm}{0.4pt}\% من المساهمين / مجلس الإدارة.
    \item للمساهمين الحق في شراء الأسهم المعروضة للبيع بالأولوية (\LR{Right of First Refusal}).
    \item تُسجّل عمليات النقل في سجل المساهمين.
\end{enumerate}

% ============================================================
% CHAPTER 7: DISSOLUTION
% ============================================================
\section{حلّ وتصفية الشركة}
\label{ch:dissolution}

\begin{enumerate}
    \item يجوز حلّ الشركة بقرار من الجمعية العامة بأغلبية \rule{2cm}{0.4pt}\%.
    \item يعيّن المصفّي من قبل الجمعية أو المحكمة.
    \item تُسدّد الديون أولاً، ثم تُوزّع الأصول المتبقية على المساهمين.
\end{enumerate}
