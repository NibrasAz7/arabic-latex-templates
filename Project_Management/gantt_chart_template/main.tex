% !TEX program = xelatex
% ============================================================
%  Arabic Gantt Chart Template — Nibras Center
%  Compile with: xelatex main.tex
% ============================================================
%  A standalone Gantt chart using TikZ (manual approach).
%  This avoids pgfgantt/bidi compatibility issues.
%  Can be included in other documents via \input{gantt.tex}
%  or compiled standalone.
% ============================================================
%  How to customize:
%    1. Change \numMonths (number of months)
%    2. Change \monthWidth (column width in cm)
%    3. Add/remove tasks in the \gantttask commands
%    4. Change colors via \definecolor
% ============================================================

\documentclass[11pt,a4paper]{article}

% ---- Geometry (landscape for wide charts) ----
\usepackage[a4paper,landscape,margin=1.5cm]{geometry}

% ---- Core packages ----
\usepackage{tikz}
\usetikzlibrary{positioning,calc,shapes.geometric}
\usepackage{xcolor}
\usepackage{fontspec}

% ---- RTL / Arabic support ----
\usepackage{bidi}
\usepackage[no-math]{fontspec}
\usepackage[quiet]{polyglossia}

% ---- Fonts ----
\setmainfont[Scale=1]{Amiri-Regular.ttf}
\newfontfamily\arabicfont[Script=Arabic,Scale=0.95]{Amiri-Regular.ttf}
\newfontfamily\englishfont[Scale=0.9]{TeX Gyre Termes}

% ---- Languages ----
\setdefaultlanguage[calendar=gregorian,hijricorrection=1]{arabic}
\setotherlanguage[variant=british]{english}

% ---- Hyperref ----
\usepackage[hidelinks]{hyperref}

% ---- Colors ----
\definecolor{ganttBar}{HTML}{2E6DA4}
\definecolor{ganttBarAlt}{HTML}{D4A017}
\definecolor{ganttMilestone}{HTML}{C0392B}
\definecolor{ganttHeader}{HTML}{1A3C6B}
\definecolor{ganttRowAlt}{HTML}{F0F4F8}
\definecolor{ganttBorder}{HTML}{AAAAAA}

% ---- Parameters ----
\newcommand{\numMonths}{12}     % total months
\newcommand{\monthWidth}{1.3}   % cm per month column
\newcommand{\taskWidth}{4.5}    % cm for task name column
\newcommand{\rowHeight}{0.8}    % cm per row
\newcommand{\headerHeight}{1.0} % cm for header row

\begin{document}
\pagestyle{empty}

\begin{center}
{\Large\bfseries مخطط جانت الزمني للمشروع}\\[1em]
\end{center}

% ============================================================
% GANTT CHART
% ============================================================
% Coordinate system:
%   x=0 is the left edge of the task name column
%   x=\taskWidth is the start of the month columns
%   Each month column is \monthWidth cm wide
%   y=0 is the top, rows go downward
%
% \gantttask{name}{startMonth}{durationMonths}{color}
%   Draws a task bar starting at startMonth (1-based) for durationMonths
% ============================================================

\begin{center}
\begin{tikzpicture}[x=1cm,y=1cm]

% ---- Calculate total width ----
\pgfmathsetmacro{\totalWidth}{\taskWidth + \numMonths*\monthWidth}

% ---- Header row: month numbers ----
\fill[ganttHeader] (0,0) rectangle (\totalWidth, \headerHeight);
\node[white,font=\bfseries] at ({\taskWidth/2}, {\headerHeight/2}) {\small المهمة};
\foreach \m in {1,...,\numMonths} {
  \pgfmathsetmacro{\xpos}{\taskWidth + (\m-0.5)*\monthWidth}
  \node[white,font=\bfseries\small] at (\xpos, {\headerHeight/2}) {\m};
}

% ---- Row backgrounds (alternating) ----
\foreach \r in {1,...,8} {
  \pgfmathsetmacro{\ytop}{-\r*\rowHeight}
  \pgfmathtruncatemacro{\isodd}{mod(\r,2)}
  \ifnum\isodd=1
    \fill[ganttRowAlt] (0,\ytop) rectangle (\totalWidth, {\ytop+\rowHeight});
  \fi
}

% ---- Horizontal grid lines ----
\foreach \r in {0,...,8} {
  \pgfmathsetmacro{\ypos}{-\r*\rowHeight}
  \draw[ganttBorder,thin] (0,\ypos) -- (\totalWidth,\ypos);
}

% ---- Vertical grid lines ----
\draw[ganttBorder,thin] (0,0) -- (0,{-8*\rowHeight});
\draw[ganttBorder,thin] (\taskWidth,0) -- (\taskWidth,{-8*\rowHeight});
\foreach \m in {1,...,\numMonths} {
  \pgfmathsetmacro{\xpos}{\taskWidth + \m*\monthWidth}
  \draw[ganttBorder,thin] (\xpos,0) -- (\xpos,{-8*\rowHeight});
}

% ---- Task names ----
\foreach \name/\r in {%
  المرحلة الأولى: المراجعة/1,%
  المرحلة الثانية: التصميم/2,%
  المرحلة الثالثة: التطوير/3,%
  المرحلة الرابعة: الاختبار/4,%
  المرحلة الخامسة: النشر/5,%
  اجتماعات المتابعة/6,%
  كتابة التقرير/7,%
  التسليم النهائي/8%
} {
  \pgfmathsetmacro{\ypos}{-\r*\rowHeight + \rowHeight/2}
  \node[anchor=west,font=\small] at (0.2,\ypos) {\name};
}

% ---- Task bars ----
% \ganttbar{row}{startMonth}{durationMonths}{color}
% Row is 1-based, months are 1-based
\newcommand{\ganttbar}[4]{%
  \pgfmathsetmacro{\ycenter}{-#1*\rowHeight + \rowHeight/2}
  \pgfmathsetmacro{\barHeight}{\rowHeight*0.5}
  \pgfmathsetmacro{\xstart}{\taskWidth + (#2-1)*\monthWidth}
  \pgfmathsetmacro{\xend}{\taskWidth + (#2-1+#3)*\monthWidth}
  \fill[#4,rounded corners=2pt] (\xstart,{\ycenter-\barHeight/2}) rectangle (\xend,{\ycenter+\barHeight/2});
}

% Task 1: months 1-3
\ganttbar{1}{1}{3}{ganttBar}
% Task 2: months 3-5
\ganttbar{2}{3}{3}{ganttBar}
% Task 3: months 5-8
\ganttbar{3}{5}{4}{ganttBar}
% Task 4: months 8-10
\ganttbar{4}{8}{3}{ganttBarAlt}
% Task 5: months 10-12
\ganttbar{5}{10}{3}{ganttBar}
% Task 6: meetings every month (bars at each month)
\foreach \m in {1,...,12} {
  \ganttbar{6}{\m}{0.3}{ganttBarAlt}
}
% Task 7: months 9-12
\ganttbar{7}{9}{4}{ganttBar}
% Task 8: month 12 (milestone)
\ganttbar{8}{12}{1}{ganttMilestone}

% ---- Milestone marker (diamond) at end of task 8 ----
\pgfmathsetmacro{\xms}{\taskWidth + 12*\monthWidth - \monthWidth/2}
\pgfmathsetmacro{\yms}{-8*\rowHeight + \rowHeight/2}
\node[diamond,fill=ganttMilestone,draw=ganttMilestone,inner sep=3pt] at (\xms,\yms) {};

% ---- Outer border ----
\draw[ganttBorder,thick] (0,0) rectangle (\totalWidth,{-8*\rowHeight});
\draw[ganttBorder,thick] (0,0) rectangle (\totalWidth,\headerHeight);

\end{tikzpicture}
\end{center}

\vspace{1.5em}
