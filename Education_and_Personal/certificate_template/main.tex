\usetikzlibrary{calc}

\newcommand{\certborder}{%
\begin{tikzpicture}[remember picture, overlay]
  % Outer frame
  \draw[certgold, line width=2.5pt]
    ($(current page.north west)+(0.8cm,-0.8cm)$)
    rectangle
    ($(current page.south east)+(-0.8cm,0.8cm)$);
  % Inner frame
  \draw[certdark, line width=1pt]
    ($(current page.north west)+(1.2cm,-1.2cm)$)
    rectangle
    ($(current page.south east)+(-1.2cm,1.2cm)$);
  % Corner ornaments (filled diamonds)
  \foreach \corner in {%
    {(0.8cm,-0.8cm)}, {(-0.8cm,-0.8cm)},
    {(0.8cm,0.8cm)}, {(-0.8cm,0.8cm)}%
  } {%
  }
  % Top-left corner
  \fill[certgold]
    ($(current page.north west)+(0.8cm,-0.8cm)$) circle (5pt);
  \fill[certgold]
    ($(current page.north east)+(-0.8cm,-0.8cm)$) circle (5pt);
  \fill[certgold]
    ($(current page.south west)+(0.8cm,0.8cm)$) circle (5pt);
  \fill[certgold]
    ($(current page.south east)+(-0.8cm,0.8cm)$) circle (5pt);
  % Decorative inner corner flourishes
  \draw[certgold, line width=0.8pt]
    ($(current page.north west)+(1.2cm,-1.2cm)$) -- ++(1.5cm,0)
    ($(current page.north west)+(1.2cm,-1.2cm)$) -- ++(0,-1.5cm);
  \draw[certgold, line width=0.8pt]
    ($(current page.north east)+(-1.2cm,-1.2cm)$) -- ++(-1.5cm,0)
    ($(current page.north east)+(-1.2cm,-1.2cm)$) -- ++(0,-1.5cm);
  \draw[certgold, line width=0.8pt]
    ($(current page.south west)+(1.2cm,1.2cm)$) -- ++(1.5cm,0)
    ($(current page.south west)+(1.2cm,1.2cm)$) -- ++(0,1.5cm);
  \draw[certgold, line width=0.8pt]
    ($(current page.south east)+(-1.2cm,1.2cm)$) -- ++(-1.5cm,0)
    ($(current page.south east)+(-1.2cm,1.2cm)$) -- ++(0,1.5cm);
\end{tikzpicture}%
}
